cmake_minimum_required(VERSION 3.24)
project(Breakly VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(breakly_core src/core.cpp src/storage.cpp)
target_include_directories(breakly_core PUBLIC src)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc /guard:cf)
  target_compile_options(breakly_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
enable_testing()
add_executable(breakly_tests tests/core_tests.cpp)
target_link_libraries(breakly_tests PRIVATE breakly_core)
add_test(NAME core COMMAND breakly_tests)
if(WIN32)
  enable_language(RC)
  add_executable(Breakly WIN32 src/main.cpp src/platform.cpp src/ui.cpp resources/Breakly.rc)
  target_compile_definitions(Breakly PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(Breakly PRIVATE breakly_core d2d1 dwrite dwmapi comctl32 uxtheme shell32 shlwapi shcore wtsapi32 powrprof advapi32 ole32 psapi)
  target_link_options(Breakly PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /guard:cf $<$<CONFIG:Release>:/OPT:REF> $<$<CONFIG:Release>:/OPT:ICF>)
  set_property(TARGET Breakly PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  install(TARGETS Breakly RUNTIME DESTINATION .)
endif()
